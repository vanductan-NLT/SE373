Chương trình model viết ở lần chạy cuối:
import os,re,collections
pat=re.compile(r'^\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2} ERROR \[service-\d+\] ([A-Za-z0-9_.\-]+):')
per=collections.defaultdict(collections.Counter)
for f in sorted(os.listdir('logs')):
    for ln in open('logs/'+f):
        m=pat.match(ln)
        if m: per[m.group(1)][f]+=1
for n in ['TimeoutError','ConnectionResetError']:
    mx=max(per[n].values())
    print(n,[ (f,c) for f,c in per[n].most_common() if c==mx])

Chương trình model viết ở lần chạy cuối:

import re, collections
files = [f'logs/service-{i:02d}.log' for i in range(1,11)]
pat = re.compile(r'ERROR \[service-\d+\] (\S+?):')
cnt = collections.Counter()
per = collections.defaultdict(collections.Counter)
err_samples=set()
lines_total=0
odd=[]
for f in files:
    for line in open(f):
        lines_total+=1
        if 'ERROR' in line:
            m = pat.search(line)
            if m:
                name=m.group(1); cnt[name]+=1; per[name][f]+=1
                if len(err_samples)<3: err_samples.add(line.strip())
            else:
                odd.append(line.strip())
print('lines', lines_total)
print('errors total matched', sum(cnt.values()))
print('top5', cnt.most_common(5))
top=cnt.most_common(1)[0]
print('winner', top, 'per-file', per[top[0]].most_common(5))
print('unmatched', len(odd), odd[:3])
print('samples', err_samples)
